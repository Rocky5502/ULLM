\begin{tikzpicture}[font=\scriptsize,>=Latex]
\tikzset{
  card/.style={draw=black!45,rounded corners=2.2pt,very thick,align=center,inner sep=4pt},
  pill/.style={draw=black!30,rounded corners=5pt,inner xsep=4pt,inner ysep=2pt,font=\scriptsize\bfseries},
  arr/.style={-{Latex[length=2mm]},thick,draw=black!60}
}
\node[card,fill=gray!6,text width=73mm] (input) at (0,0) {
  \textbf{Imperfective telic evidence}\\[-1pt]
  \emph{Premise:} ``The carpenter was building a gazebo.''\\
  \emph{Hypothesis:} ``The carpenter built a gazebo.''
};
\node[card,fill=blue!7,text width=33mm] (semantic) at (-20mm,-18mm) {
  \textbf{Event-level semantic uncertainty}\\
  Did the real-world endpoint occur?\\[1pt]
  \textbf{Under-specified}
};
\node[card,fill=orange!8,text width=33mm] (predictive) at (20mm,-18mm) {
  \textbf{Model-level predictive uncertainty}\\
  How dispersed is belief over\\ $\{T,F,U\}$?
};
\draw[arr] (input.south west) ++(12mm,0) -- (semantic.north);
\draw[arr] (input.south east) ++(-12mm,0) -- (predictive.north);
\node[card,fill=green!8,text width=22mm,minimum height=22mm] (ideal) at (-28mm,-47mm) {
  \textbf{Desired}\\
  $P(U)\!\gg\!P(T),P(F)$\\
  \textbf{confidently uncertain}
};
\node[card,fill=red!7,text width=22mm,minimum height=22mm] (teleo) at (0,-47mm) {
  \textbf{Teleological error}\\
  $P(T)\!\gg\!P(U),P(F)$\\
  \textbf{false completion}
};
\node[card,fill=yellow!12,text width=22mm,minimum height=22mm] (confused) at (28mm,-47mm) {
  \textbf{Predictive confusion}\\
  $P(T)\!\approx\!P(F)\!\approx\!P(U)$\\
  \textbf{high entropy}
};
\draw[arr] (semantic.south) -- (ideal.north);
\draw[arr] (predictive.south) -- (teleo.north);
\draw[arr] (predictive.south) -- (confused.north);
\node[pill,fill=purple!7,text width=70mm,align=center] (takeaway) at (0,-67mm) {
  Group C has an uncertain \emph{world state} but a determinate NLI target: \textsc{Unknown}.\\
  Therefore high entropy is \emph{not} the objective; high calibrated $P(\textsc{Unknown})$ is.
};
\draw[arr] (ideal.south) |- (takeaway.west);
\draw[arr] (teleo.south) -- (takeaway.north);
\draw[arr] (confused.south) |- (takeaway.east);
\end{tikzpicture}
